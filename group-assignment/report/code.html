  State<FormWizardPage> createState() => _FormWizardPageState();

class _FormWizardPageState extends State<FormWizardPage> {
  final List<String> stepTitles = [
    'Personal',
    'Contact',
    'Preferences',
    'Review',
  ];
  int currentStep = 0;

  // One form key for each step that has inputs (steps 1-3).
  final List<GlobalKey<FormState>> formKeys = [
    GlobalKey<FormState>(),
    GlobalKey<FormState>(),
    GlobalKey<FormState>(),
  ];

  // Step 1: Personal Information
  final nameController = TextEditingController();
  final birthDateController = TextEditingController();
  DateTime? birthDate;
  String? gender;

  // Step 2: Contact Details
  final emailController = TextEditingController();
  final phoneController = TextEditingController();
  final addressController = TextEditingController();

  // Step 3: Preferences & Settings
  final countryController = TextEditingController();
  final List<String> languages = ['English', 'Bangla', 'Hindi', 'Arabic'];
  String? language;
  bool notifications = true;
  bool newsletter = false;

  @override
  void initState() {
    super.initState();
    if (DraftStorage.hasDraft) {
      loadDraft();
      // Show the message after the first frame is drawn.
      WidgetsBinding.instance.addPostFrameCallback((_) {
        showMessage('Draft restored — continue where you left off');
      });
    }
  }

  @override
  void dispose() {
    nameController.dispose();
    birthDateController.dispose();
    emailController.dispose();
    phoneController.dispose();
    addressController.dispose();
    countryController.dispose();
    super.dispose();
  }

  // ---------- Draft ----------

  void saveDraft() {
    DraftStorage.save({
      'step': currentStep,
      'name': nameController.text,
      'birthDate': birthDate,
      'gender': gender,
      'email': emailController.text,
      'phone': phoneController.text,
      'address': addressController.text,
      'country': countryController.text,
      'language': language,
      'notifications': notifications,
      'newsletter': newsletter,
    });
    showMessage('Draft saved ✔');
  }

  void loadDraft() {
    final draft = DraftStorage.load()!;
    currentStep = draft['step'];
    nameController.text = draft['name'];
    birthDate = draft['birthDate'];
    if (birthDate != null) {
      birthDateController.text = formatDate(birthDate!);
    }
    gender = draft['gender'];
    emailController.text = draft['email'];
    phoneController.text = draft['phone'];
    addressController.text = draft['address'];
    countryController.text = draft['country'];
    language = draft['language'];
    notifications = draft['notifications'];
    newsletter = draft['newsletter'];
  }

  void showMessage(String text) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(text), duration: const Duration(seconds: 2)),
    );
  }

  // ---------- Navigation ----------

  void goNext() {
    // Only move forward when the current step is valid.
    if (formKeys[currentStep].currentState!.validate()) {
      setState(() => currentStep++);
    } else {
      showMessage('Please fix the errors before continuing');
    }
  }

  void goBack() {
    setState(() => currentStep--);
  }

  void submit() {
    DraftStorage.clear();
    showDialog(
      context: context,
      barrierDismissible: false,
      builder: (context) => AlertDialog(
        title: const Text('Submitted Successfully 🎉'),
        content: Text('Thank you, ${nameController.text}!'),

          TextButton(
            onPressed: () {
              Navigator.pop(context); // close the dialog
              Navigator.pop(this.context); // back to the home page
            },
            child: const Text('Done'),
          ),
        ],
      ),
    );
  }

// All form validation rules in one place.
// Each method returns an error message, or null when the value is valid.
class Validators {
  static String? nameValidator(String? name) {
    if (name == null || name.trim().isEmpty) {
      return 'Name is required';
    }
    if (name.trim().length < 2) {
      return 'Name must be at least 2 characters long';
    }
    if (!RegExp(r'^[a-zA-Z\s.]+$').hasMatch(name.trim())) {
      return 'Name can only contain letters and spaces';
    }
    return null;
  }

  static String? emailValidation(String? email) {
    if (email == null || email.trim().isEmpty) {
      return 'Email is required';
    }
    final emailRegex = RegExp(r'^[\w.-]+@([\w-]+\.)+[\w-]{2,4}$');
    if (!emailRegex.hasMatch(email.trim())) {
      return 'Please enter a valid email address';
    }
    return null;
  }

  // Accepts 01XXXXXXXXX, +8801XXXXXXXXX or 8801XXXXXXXXX
  static String? bangladeshNumberValidator(String? phone) {
    if (phone == null || phone.trim().isEmpty) {
      return 'Phone number is required';
    }
    String cleanPhone = phone.replaceAll(RegExp(r'[^\d]'), '');
    if (cleanPhone.startsWith('880')) {
      cleanPhone = cleanPhone.substring(2); // keep the leading 0
    }
    if (cleanPhone.length != 11) {
      return 'Phone number must be 11 digits (e.g. 01712345678)';
    }
    const validPrefixes = ['013', '014', '015', '016', '017', '018', '019'];
    if (!validPrefixes.contains(cleanPhone.substring(0, 3))) {
      return 'Please enter a valid Bangladesh mobile number';
    }
    return null;
  }

  static String? ageValidator(String? age) {
    if (age == null || age.trim().isEmpty) {
      return 'Age is required';
    }
    final int? ageValue = int.tryParse(age.trim());
    if (ageValue == null) {
      return 'Please enter a valid age';
    }
    if (ageValue < 13) {
      return 'You must be at least 13 years old';
    }
    if (ageValue > 120) {
      return 'Please enter a valid age';
    }
    return null;
  }

  // Bangladesh NID is 10, 13 or 17 digits
  static String? nidValidator(String? nid) {
    if (nid == null || nid.trim().isEmpty) {
      return 'NID is required';
    }
    final String cleanNid = nid.replaceAll(RegExp(r'[^\d]'), '');
    if (cleanNid.length != 10 &&
        cleanNid.length != 13 &&
        cleanNid.length != 17) {
      return 'NID must be 10, 13 or 17 digits';
    }
    return null;
  }

  static String? passwordValidator(String? password) {
    if (password == null || password.isEmpty) {
      return 'Password is required';
    }
    if (password.length < 8) {
      return 'Password must be at least 8 characters long';
    }
    if (!password.contains(RegExp(r'[A-Z]'))) {
      return 'Password must contain at least one uppercase letter';
    }
    if (!password.contains(RegExp(r'[a-z]'))) {
      return 'Password must contain at least one lowercase letter';
    }
    if (!password.contains(RegExp(r'[0-9]'))) {
      return 'Password must contain at least one number';
    }
    if (!password.contains(RegExp(r'[!@#$%^&*(),.?":{}|<>_\-]'))) {
      return 'Password must contain at least one special character';
    }
    return null;
  }

  static String? confirmPasswordValidator(
    String? password,
    String? confirmPassword,
  ) {
    if (confirmPassword == null || confirmPassword.isEmpty) {
      return 'Please confirm your password';
    }
    if (password != confirmPassword) {
      return 'Passwords do not match';
    }
    return null;
  }

  static String? requiredValidator(String? value, String fieldName) {
    if (value == null || value.trim().isEmpty) {
      return '$fieldName is required';
    }
    return null;
  }

  static String? minLengthValidator(
    String? value,
    int minLength,
    String fieldName,
  ) {
    if (value == null || value.trim().isEmpty) {
      return '$fieldName is required';
    }
    if (value.trim().length < minLength) {
      return '$fieldName must be at least $minLength characters long';
    }
    return null;
  }

  // Password strength from 0 (empty) to 5 (meets every rule)
  static int passwordStrength(String password) {
    int score = 0;
    if (password.length >= 8) score++;
    if (password.contains(RegExp(r'[A-Z]'))) score++;
    if (password.contains(RegExp(r'[a-z]'))) score++;
    if (password.contains(RegExp(r'[0-9]'))) score++;
    if (password.contains(RegExp(r'[!@#$%^&*(),.?":{}|<>_\-]'))) score++;
    return score;
  }
}

// Small helpers for showing and checking dates without extra packages.
String formatDate(DateTime date) {
  final String day = date.day.toString().padLeft(2, '0');
  final String month = date.month.toString().padLeft(2, '0');
  return '$day/$month/${date.year}';
}

int ageFromBirthDate(DateTime birthDate) {
  final DateTime today = DateTime.now();
  int age = today.year - birthDate.year;
  final bool birthdayNotYet =
      today.month < birthDate.month ||
      (today.month == birthDate.month && today.day < birthDate.day);
  if (birthdayNotYet) {
    age--;
  }
  return age;
}

import '../utils/validators.dart';

// Password field with a show/hide button and an optional strength bar.
class PasswordField extends StatefulWidget {
  final TextEditingController controller;
  final String labelText;
  final bool showStrength;
  final String? Function(String?)? validator;

  const PasswordField({
    super.key,
    required this.controller,
    this.labelText = 'Password',
    this.showStrength = false,
    this.validator,
  });

  @override
  State<PasswordField> createState() => _PasswordFieldState();
}

class _PasswordFieldState extends State<PasswordField> {
  bool _obscureText = true;

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 16),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          TextFormField(
            controller: widget.controller,
            obscureText: _obscureText,
            autovalidateMode: AutovalidateMode.onUserInteraction,
            onChanged: (_) => setState(() {}), // redraw the strength bar
            decoration: InputDecoration(
              labelText: widget.labelText,
              prefixIcon: const Icon(Icons.lock),
              suffixIcon: IconButton(
                icon: Icon(
                  _obscureText ? Icons.visibility : Icons.visibility_off,
                ),
                onPressed: () {
                  setState(() {
                    _obscureText = !_obscureText;
                  });
                },
              ),
              border: OutlineInputBorder(
                borderRadius: BorderRadius.circular(10),
              ),
            ),
            validator: widget.validator,
          ),
          if (widget.showStrength) _buildStrengthBar(),
        ],
      ),
    );
  }

  Widget _buildStrengthBar() {
    final int score = Validators.passwordStrength(widget.controller.text);

    String label = 'Too weak';
    Color color = Colors.red;
    if (widget.controller.text.isEmpty) {
      label = '-';
      color = Colors.grey;
    } else if (score >= 5) {
      label = 'Strong';
      color = Colors.green;
    } else if (score >= 3) {
      label = 'Medium';
      color = Colors.orange;
    }

    return Padding(
      padding: const EdgeInsets.only(top: 8, left: 4, right: 4),
      child: Row(
        children: [
          Expanded(
            child: LinearProgressIndicator(
              value: score / 5,
              minHeight: 6,
              color: color,
              backgroundColor: Colors.grey.shade300,
              borderRadius: BorderRadius.circular(4),
            ),
          ),
          const SizedBox(width: 12),
          Text(
            'Strength: $label',
            style: TextStyle(color: color, fontWeight: FontWeight.bold),
          ),
        ],
      ),
    );
  }
}
